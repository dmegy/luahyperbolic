\begin{luacode*}
local A, B, C = 1/3, -1/5+complex.I/2, -1/2-complex.I/5

local AA = hyper.symmetry(A)(C)
local BB = hyper.symmetry(B)(C)

local e1, e2 = hyper.endpointsAngleBisector(AA,A,B)
local f1, f2 = hyper.endpointsAngleBisector(A,B,BB)
local I_C = hyper.interLL(e1, e2, f1, f2) -- can be nil !

hyper.tikzBegin("scale=6")
hyper.drawLine(C,A,"gray,dashed")
hyper.drawLine(C,B,"gray,dashed")
hyper.drawLine(e1,e2,"red")
hyper.drawLine(f1,f2,"red")
hyper.drawTriangle(A,B,C,"very thick, black")
hyper.tikzEnd()
\end{luacode*}
